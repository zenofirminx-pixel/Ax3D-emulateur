cmake_minimum_required(VERSION 3.22.1)
project(ax3d LANGUAGES CXX)
add_library(ax3d SHARED native-lib.cpp core.cpp iso_reader.cpp sfo.cpp elf_loader.cpp cpu/allegrex.cpp cso_reader.cpp abx_vm.cpp)
